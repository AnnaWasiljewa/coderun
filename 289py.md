[Проверка палиндрома](https://coderun.yandex.ru/problem/palindroming-check/description)


```
def normalize(s:str) -> str:
    return "".join(ch.lower() for ch in s if ch.isalpha())

def is_palindrom(s:str) -> bool:
    left, right = 0, len(s) - 1
    while left < right:
        if s[left] != s[right]:
            return False
        left += 1
        right -= 1
    return True

def main():
    raw = input()
    clean = normalize(raw)
    if is_palindrom(clean):
        print("It is a palindrome")
    else:
        print("It is not a palindrome")


if __name__ == '__main__':
    main()
```
