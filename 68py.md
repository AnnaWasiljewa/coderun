[Полиглоты](https://coderun.yandex.ru/problem/polyglots/description)
```py
n = int(input())
all_lang = set()
one_pupil = None

for _ in range(n):
    m = int(input())
    st = set()
    for i in range(m):
        lang = input()
        st.add(lang)
    all_lang |= st
    if one_pupil is None:
        one_pupil = st.copy()
    else:
        one_pupil &= st

print(len(one_pupil))
for lang in sorted(one_pupil):
    print(lang)
print(len(all_lang))
for lang in sorted(all_lang):
    print(lang)

```
